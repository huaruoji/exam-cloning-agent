\section{核心技术创新}
% ============================================================

\begin{frame}{创新一：资源感知 LLM 路由}
  \vspace{1mm}
  \begin{tcolorbox}[coralbox, title=六级路由优先级（算力网核心）]
    \scriptsize
    \begin{enumerate}\setlength\itemsep{0.2mm}
      \item \textbf{规则批改}：MCQ / 判断题本地确定性判定（零算力）
      \item \textbf{题库复用}：命中已存题目直接返回（零算力）
      \item \textbf{用户选定 provider}：用户配置的健康端点优先
      \item \textbf{管理员本地 provider}：Ollama / vLLM 等内网服务
      \item \textbf{内置 provider}：DeepSeek 兜底（允许 fallback 时）
      \item \textbf{显式降级}：无可用 provider 时返回未批改，不误判
    \end{enumerate}
  \end{tcolorbox}
  \vspace{2mm}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{\color{ink}容错机制}
    \begin{itemize}\scriptsize\setlength\itemsep{0.2mm}
      \item 每-provider 并发上限 + 超时
      \item 3 次连续失败 → 熔断 60 秒
      \item 探测成功 → 闭合熔断器
      \item 重试有界，避免重复计费
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{\color{ink}安全与隐私}
    \begin{itemize}\scriptsize\setlength\itemsep{0.2mm}
      \item 用户密钥仅存浏览器，请求级附带
      \item 用户 URL 禁止内网/回环（防 SSRF）
      \item 管理员本地端点经环境变量配置
      \item 遥测不记录密钥与完整 prompt
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{创新二：SM-2 自适应学习引擎}
  \vspace{1mm}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{tcolorbox}[claudecard, title=算法核心]
      \scriptsize
      \textbf{SM-2 间隔重复}：基于连续答对次数与难度，动态调度复习间隔。\\[0.5mm]
      \textbf{持久化 ease\_factor}：每个知识点维护独立难度系数。\\[0.5mm]
      \textbf{Beta 先验}：冷启动用 Beta 分布平滑初始评分。\\[0.5mm]
      \textbf{due-concept 调度}：优先推送到期薄弱知识点。
    \end{tcolorbox}
    \column{0.46\textwidth}
    \begin{tcolorbox}[claudecard, title=数据流]
      \scriptsize
      \begin{enumerate}\setlength\itemsep{0.2mm}
        \item 答题 → LLM 批改 → 更新掌握度
        \item 误答 → 加入错题本
        \item 错题 → 间隔重复调度
        \item 模拟考错题 → 一键导入练习
      \end{enumerate}
    \end{tcolorbox}
  \end{columns}
  \vspace{2mm}
  \begin{center}
  \begin{tcolorbox}[claudebox, width=0.86\textwidth]
    \centering\scriptsize\color{ink}
    \textbf{效果}：客观题零算力即时反馈 \;·\; 主观题 LLM 深度批改 \;·\; 薄弱点自动浮现 \;·\; 复习间隔科学化
  \end{tcolorbox}
  \end{center}
\end{frame}

\begin{frame}{创新三：考试风格克隆与全题型批改}
  \vspace{1mm}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{\color{ink}考试风格克隆}
    \begin{tcolorbox}[claudebox]
      \scriptsize
      \begin{itemize}\setlength\itemsep{0.2mm}
        \item pdfplumber 提取 + LLM 结构化解析
        \item 从试卷/作业聚合风格画像
        \item 生成时 few-shot 注入 3 道参考题
        \item 题型/难度分布可调（滑块）
        \item 题库/AI 生成比默认 80/20
      \end{itemize}
    \end{tcolorbox}
    \column{0.48\textwidth}
    \textbf{\color{ink}全题型 LLM 批改}
    \begin{tcolorbox}[claudebox]
      \scriptsize
      支持五种题型：
      \begin{itemize}\setlength\itemsep{0.2mm}
        \item 选择题 / 判断题（规则批改）
        \item 简答题 / 计算题 / 论述题（LLM）
      \end{itemize}
      结构化反馈：
      \begin{itemize}\setlength\itemsep{0.2mm}
        \item \textcolor{coral}{correct} 是否正确
        \item \textcolor{coral}{missing\_steps} 缺失步骤
        \item \textcolor{coral}{wrong\_concepts} 错误概念
        \item \textcolor{coral}{suggestion} 改进建议
      \end{itemize}
    \end{tcolorbox}
  \end{columns}
\end{frame}
